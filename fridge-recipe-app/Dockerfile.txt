# 1. Node.js 기반의 의존성 설치 단계
FROM node:18-alpine AS deps             # 가벼운 Alpine 기반 Node 18 이미지 사용 (의존성 설치용)
WORKDIR /app                            # 작업 디렉터리 설정
COPY package.json package-lock.json ./  # 의존성 설치에 필요한 파일 복사
RUN npm ci                              # CI 환경에 적합한 패키지 설치 방식 (clean install)

# 2. 애플리케이션 빌드 단계
FROM node:18-alpine AS builder          # 빌드용 이미지 (코드 포함 전체 복사 및 빌드)
WORKDIR /app
COPY . .                                # 모든 소스 코드 복사
RUN npm run build                       # 앱 빌드 실행

# 3. 실제 실행용 런타임 이미지
FROM node:18-alpine AS runner           # 최종 실행 환경
WORKDIR /app
ENV NODE_ENV=production                 # 환경변수 설정: production 모드
# 빌드 결과물 및 필요한 파일만 복사해서 이미지 용량 최소화
COPY --from=builder /app/public ./public
COPY --from=builder /app/.next ./.next
COPY --from=deps /app/node_modules ./node_modules
COPY --from=builder /app/package.json ./package.json

EXPOSE 3000                             # 외부에서 접속 가능한 포트 지정 (Next.js 기본 포트)
CMD ["npm", "run", "start"]             # 컨테이너 시작 시 실행될 명령어 (Next.js 앱 실행)
